1. Core Web Vitals Overview

Core Web Vitals are a set of real-world user experience metrics that measure key aspects of user experience. They focus on three areas: loading, interactivity, and visual stability.

Three Core Web Vitals:

  • Largest Contentful Paint (LCP): Measures perceived loading speed. Target: < 2.5 seconds
  • First Input Delay (FID) / Interaction to Next Paint (INP): Measures interactivity. Target: < 100ms
  • Cumulative Layout Shift (CLS): Measures visual stability. Target: < 0.1
Key Rule: Performance is not just about speed - it's about perceived performance and user experience. Fast loading times and smooth interactions keep users engaged.

2. Loading Performance Optimization

Optimize how your portfolio loads to ensure users can start interacting quickly.

Loading Optimization Strategies:

  • Critical CSS Inlining: Inline above-the-fold CSS for faster initial render
  • Resource Hints: Use preconnect, prefetch, and preload for better resource loading
  • Code Splitting: Split JavaScript into smaller, focused chunks
  • Lazy Loading: Load images and components only when needed
  • Caching Strategies: Implement proper browser and CDN caching

Critical CSS Implementation:

// critical-css.js
class CriticalCSSOptimizer {
    constructor() {
        this.criticalCSS = `
            /* Above-the-fold styles */
            * { margin: 0; padding: 0; box-sizing: border-box; }
            :root {
                --primary: #d90429;
                --secondary: #ef233c;
                --background: #fcf7f7;
                --text: #1c0b0f;
            }
            body {
                font-family: Inter, system-ui, sans-serif;
                background: var(--background);
                color: var(--text);
                line-height: 1.6;
            }
            .hero {
                min-height: 60vh;
                display: flex;
                align-items: center;
                justify-content: center;
                text-align: center;
                padding: 2rem;
            }
            .hero h1 {
                font-size: 2.5rem;
                margin-bottom: 1rem;
                color: var(--primary);
            }
        `;
        this.init();
    }

    init() {
        this.injectCriticalCSS();
        this.loadAsyncCSS();
    }

    injectCriticalCSS() {
        const style = document.createElement('style');
        style.textContent = this.criticalCSS;
        document.head.appendChild(style);
    }

    loadAsyncCSS() {
        const nonCriticalCSS = [
           ../../../Assets/css/main.css',
           ../../../Assets/css/responsive.css'
        ];

        nonCriticalCSS.forEach(url => {
            const link = document.createElement('link');
            link.rel = 'preload';
            link.href = url;
            link.as = 'style';
            link.onload = () => {
                link.rel = 'stylesheet';
            };
            document.head.appendChild(link);
        });
    }
}

3. Image Optimization

Images often account for the largest portion of web page weight. Optimize them for both quality and performance.

Image Optimization Techniques:

  • Modern Formats: Use WebP, AVIF, or JPEG 2000 for better compression
  • Responsive Images: Serve appropriately sized images for different devices
  • Lazy Loading: Load images only when they enter the viewport
  • CDN Optimization: Use CDN with image optimization services
  • Compression: Optimize file size without sacrificing quality

Image Optimization Implementation:

// image-optimization.js
class ImageOptimizer {
    constructor() {
        this.lazyLoadThreshold = 200;
        this.init();
    }

    init() {
        this.setupLazyLoading();
        this.setupResponsiveImages();
        this.optimizeExistingImages();
    }

    setupLazyLoading() {
        const images = document.querySelectorAll('img[data-src]');
        
        const imageObserver = new IntersectionObserver((entries, observer) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const img = entry.target;
                    img.src = img.dataset.src;
                    img.classList.add('lazy-loaded');
                    observer.unobserve(img);
                }
            });
        }, {
            rootMargin: `${this.lazyLoadThreshold}px 0px`
        });

        images.forEach(img => imageObserver.observe(img));
    }

    setupResponsiveImages() {
        const pictures = document.querySelectorAll('picture');
        
        pictures.forEach(picture => {
            const sources = picture.querySelectorAll('source');
            sources.forEach(source => {
                const sizes = source.getAttribute('data-sizes') || '(max-width: 768px) 100vw, 50vw';
                const srcset = source.getAttribute('data-srcset');
                
                if (srcset) {
                    source.setAttribute('srcset', srcset);
                    source.setAttribute('sizes', sizes);
                }
            });
        });
    }

    optimizeExistingImages() {
        const images = document.querySelectorAll('img');
        
        images.forEach(img => {
            // Convert to WebP if supported
            if (this.supportsWebP()) {
                const webpSrc = img.src.replace(/\.(jpg|jpeg|png)$/, '.webp');
                img.srcset = `${webpSrc} 1x, ${img.src} 2x`;
                img.src = webpSrc;
            }

            // Add loading attribute
            img.loading = 'lazy';
            
            // Add error handling
            img.onerror = () => {
                img.src = '/assets/images/fallback.jpg';
            };
        });
    }

    supportsWebP() {
        const canvas = document.createElement('canvas');
        return canvas.toDataURL('image/webp').indexOf('data:image/webp') === 0;
    }
}

4. JavaScript Performance

Optimize JavaScript execution to improve interactivity and reduce main thread blocking.

JavaScript Optimization Strategies:

  • Code Splitting: Split code into chunks that load on demand
  • Tree Shaking: Remove unused code from bundles
  • Lazy Loading Components: Load components only when needed
  • Debouncing and Throttling: Optimize event handlers
  • Web Workers: Offload heavy processing to background threads

JavaScript Optimization Implementation:

// javascript-optimization.js
class JavaScriptOptimizer {
    constructor() {
        this.init();
    }

    init() {
        this.setupCodeSplitting();
        this.setupDebouncing();
        this.setupLazyComponents();
        this.setupWebWorkers();
    }

    setupCodeSplitting() {
        // Dynamic imports for non-critical modules
        const loadNonCriticalModules = async () => {
            const modules = [
                './components/Analytics.js',
                './components/Animations.js',
                './components/AdvancedFeatures.js'
            ];

            modules.forEach(module => {
                import(module).then(module => {
                    // Initialize module
                    if (module.init) {
                        module.init();
                    }
                }).catch(error => {
                    console.warn(`Failed to load module: ${module}`, error);
                });
            });
        };

        // Load non-critical modules after page load
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', loadNonCriticalModules);
        } else {
            setTimeout(loadNonCriticalModules, 1000);
        }
    }

    setupDebouncing() {
        const debouncedFunctions = new Map();

        const debounce = (func, wait) => {
            return (...args) => {
                const context = this;
                const later = () => {
                    func.apply(context, args);
                };
                clearTimeout(debouncedFunctions.get(func));
                debouncedFunctions.set(func, setTimeout(later, wait));
            };
        };

        // Apply debouncing to scroll events
        const handleScroll = debounce(() => {
            this.handleScrollEvent();
        }, 100);

        window.addEventListener('scroll', handleScroll);
    }

    setupLazyComponents() {
        const lazyComponents = document.querySelectorAll('[data-lazy-component]');

        const componentObserver = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    const componentName = entry.target.dataset.lazyComponent;
                    this.loadComponent(componentName);
                    componentObserver.unobserve(entry.target);
                }
            });
        }, { rootMargin: '50px' });

        lazyComponents.forEach(component => {
            componentObserver.observe(component);
        });
    }

    loadComponent(componentName) {
        import(`./components/${componentName}.js`)
            .then(module => {
                if (module.default) {
                    module.default();
                }
            })
            .catch(error => {
                console.warn(`Failed to load component: ${componentName}`, error);
            });
    }

    setupWebWorkers() {
        // Create Web Worker for heavy processing
        const workerCode = `
            self.onmessage = function(e) {
                const { data } = e;
                const result = heavyProcessing(data);
                self.postMessage(result);
            };

            function heavyProcessing(data) {
                // Perform CPU-intensive operations
                return data.map(item => {
                    return item * 2; // Example processing
                });
            }
        `;

        const blob = new Blob([workerCode], { type: 'application/javascript' });
        const worker = new Worker(URL.createObjectURL(blob));

        // Use worker for heavy tasks
        this.processWithWorker = (data) => {
            return new Promise((resolve) => {
                worker.onmessage = (e) => {
                    resolve(e.data);
                };
                worker.postMessage(data);
            });
        };
    }

    async handleScrollEvent() {
        // Handle scroll events efficiently
        const scrollPosition = window.scrollY;
        const windowHeight = window.innerHeight;
        const documentHeight = document.documentElement.scrollHeight;

        // Calculate scroll percentage
        const scrollPercentage = (scrollPosition / (documentHeight - windowHeight)) * 100;

        // Update progress indicator
        this.updateScrollProgress(scrollPercentage);
    }

    updateScrollProgress(percentage) {
        const progressBars = document.querySelectorAll('.scroll-progress');
        progressBars.forEach(bar => {
            bar.style.width = `${percentage}%`;
        });
    }
}

5. Performance Monitoring

Set up ongoing performance monitoring to maintain optimal user experience.

Monitoring Strategies:

  • Real User Monitoring (RUM): Collect performance data from real users
  • Core Web Vitals Tracking: Monitor LCP, INP, and CLS over time
  • Error Tracking: Monitor JavaScript errors and performance issues
  • CI/CD Integration: Include performance tests in build process
  • Alerting: Set up alerts for performance regressions

Performance Monitoring Implementation:

// performance-monitoring.js
class PerformanceMonitor {
    constructor() {
        this.metrics = {
            lcp: null,
            fid: null,
            cls: null,
            navigation: null
        };
        this.init();
    }

    init() {
        this.setupCoreWebVitals();
        this.setupCustomMetrics();
        this.setupReporting();
    }

    setupCoreWebVitals() {
        // Largest Contentful Paint
        new PerformanceObserver((entryList) => {
            const entries = entryList.getEntries();
            const lastEntry = entries[entries.length - 1];
            this.metrics.lcp = lastEntry.startTime;
            this.reportMetric('lcp', lastEntry.startTime);
        }).observe({ entryTypes: ['largest-contentful-paint'] });

        // First Input Delay
        new PerformanceObserver((entryList) => {
            for (const entry of entryList.getEntries()) {
                this.metrics.fid = entry.processingStart - entry.startTime;
                this.reportMetric('fid', this.metrics.fid);
            }
        }).observe({ entryTypes: ['first-input'] });

        // Cumulative Layout Shift
        let clsValue = 0;
        new PerformanceObserver((entryList) => {
            for (const entry of entryList.getEntries()) {
                if (!entry.hadRecentInput) {
                    clsValue += entry.value;
                    this.metrics.cls = clsValue;
                    this.reportMetric('cls', clsValue);
                }
            }
        }).observe({ entryTypes: ['layout-shift'] });
    }

    setupCustomMetrics() {
        // Navigation timing
        window.addEventListener('load', () => {
            const navigation = performance.getEntriesByType('navigation')[0];
            this.metrics.navigation = {
                dns: navigation.domainLookupEnd - navigation.domainLookupStart,
                tcp: navigation.connectEnd - navigation.connectStart,
                request: navigation.responseStart - navigation.requestStart,
                response: navigation.responseEnd - navigation.responseStart,
                dom: navigation.domInteractive - navigation.responseEnd,
                load: navigation.loadEventEnd - navigation.domContentLoadedEventStart
            };
            this.reportMetric('navigation', this.metrics.navigation);
        });

        // Custom timing for portfolio-specific interactions
        this.trackInteractionTiming();
    }

    trackInteractionTiming() {
        const interactionStart = new Map();

        document.addEventListener('mousedown', (e) => {
            interactionStart.set(e.target, performance.now());
        });

        document.addEventListener('mouseup', (e) => {
            const startTime = interactionStart.get(e.target);
            if (startTime) {
                const duration = performance.now() - startTime;
                this.reportMetric('interaction', {
                    element: e.target.tagName.toLowerCase(),
                    duration: duration
                });
                interactionStart.delete(e.target);
            }
        });
    }

    setupReporting() {
        // Report to analytics service
        setInterval(() => {
            this.sendToAnalytics();
        }, 30000); // Report every 30 seconds

        // Report on page unload
        window.addEventListener('beforeunload', () => {
            this.sendToAnalytics();
        });
    }

    reportMetric(name, value) {
        console.log(`Performance Metric - ${name}:`, value);
        
        // Send to analytics service
        if (window.gtag) {
            window.gtag('event', 'performance_metric', {
                metric_name: name,
                metric_value: value,
                page_path: window.location.pathname
            });
        }

        // Store for local reporting
        this.metrics[name] = value;
    }

    sendToAnalytics() {
        const payload = {
            timestamp: new Date().toISOString(),
            url: window.location.href,
            metrics: this.metrics,
            userAgent: navigator.userAgent,
            screen: {
                width: screen.width,
                height: screen.height
            }
        };

        // Send to your analytics endpoint
        fetch('/api/performance', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify(payload)
        }).catch(error => {
            console.warn('Failed to send performance data:', error);
        });
    }

    getPerformanceScore() {
        const scores = {
            lcp: this.getLCPScore(),
            fid: this.getFIDScore(),
            cls: this.getCLSScore(),
            navigation: this.getNavigationScore()
        };

        const overall = (scores.lcp + scores.fid + scores.cls + scores.navigation) / 4;
        return {
            overall: Math.round(overall),
            breakdown: scores
        };
    }

    getLCPScore() {
        if (!this.metrics.lcp) return 0;
        if (this.metrics.lcp < 2500) return 100;
        if (this.metrics.lcp < 4000) return 75;
        if (this.metrics.lcp < 7000) return 50;
        return 0;
    }

    getFIDScore() {
        if (!this.metrics.fid) return 0;
        if (this.metrics.fid < 100) return 100;
        if (this.metrics.fid < 300) return 75;
        if (this.metrics.fid < 500) return 50;
        return 0;
    }

    getCLSScore() {
        if (!this.metrics.cls) return 100;
        if (this.metrics.cls < 0.1) return 100;
        if (this.metrics.cls < 0.25) return 75;
        if (this.metrics.cls < 0.5) return 50;
        return 0;
    }

    getNavigationScore() {
        if (!this.metrics.navigation) return 0;
        const total = Object.values(this.metrics.navigation).reduce((a, b) => a + b, 0);
        if (total < 3000) return 100;
        if (total < 5000) return 75;
        if (total < 7000) return 50;
        return 0;
    }
}
